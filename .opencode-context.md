## Component Template

Use this template when creating new components:

```typescript
'use client';

import React, { ComponentPropsWithoutRef } from 'react';

interface ComponentNameProps extends ComponentPropsWithoutRef<'div'> {
  variant?: 'primary' | 'secondary' | 'ghost';
  size?: 'sm' | 'md' | 'lg';
  disabled?: boolean;
  children: React.ReactNode;
}

/**
 * ComponentName - Brief description
 * 
 * @example
 * <ComponentName variant="primary" size="md">
 *   Content
 * </ComponentName>
 */
export const ComponentName: React.FC<ComponentNameProps> = ({
  variant = 'primary',
  size = 'md',
  disabled = false,
  children,
  className,
  ...props
}) => {
  return (
    <div
      className={`
        // Base styles
        bg-white rounded-base transition-all duration-base
        
        // Variant styles
        ${variant === 'primary' && 'bg-primary text-white'}
        ${variant === 'secondary' && 'bg-grey-light text-grey'}
        ${variant === 'ghost' && 'bg-transparent text-primary'}
        
        // Size styles
        ${size === 'sm' && 'px-3 py-1 text-sm'}
        ${size === 'md' && 'px-4 py-2 text-base'}
        ${size === 'lg' && 'px-6 py-3 text-lg'}
        
        // Disabled state
        ${disabled && 'opacity-50 cursor-not-allowed'}
        
        // Custom classes
        ${className}
      `}
      {...props}
    >
      {children}
    </div>
  );
};

export default ComponentName;
```

## CSS Best Practices

1. **Use Tailwind Utilities**
   ```
   ✓ className="bg-primary text-white rounded-base"
   ✗ className="background-color: #5B9BD5; color: white;"
   ```

2. **Mobile-First**
   ```
   ✓ className="text-sm md:text-base lg:text-lg"
   ✗ className="lg:text-sm" (without mobile fallback)
   ```

3. **Use Design Tokens**
   ```
   ✓ className="text-primary hover:text-primary-dark"
   ✗ className="text-#5B9BD5"
   ```

4. **Organize Classes**
   ```
   Order: Layout → Spacing → Colors → Typography → Interactive
   
   ✓ flex gap-4 p-6 bg-white text-grey text-base hover:text-primary
   ✗ text-base hover:text-primary p-6 flex gap-4 bg-white text-grey
   ```

## Naming Conventions

### Files
```
Components: PascalCase
  Button.tsx, QuoteForm.tsx, ServiceCard.tsx

Utilities: camelCase
  formatDate.ts, calculateTotal.ts

Styles: kebab-case
  globals.css, variables.css
```

### Variables
```
React Components: PascalCase
  const MyComponent = () => {}

Functions: camelCase
  const calculatePrice = () => {}

Constants: SCREAMING_SNAKE_CASE
  const MAX_QUOTES = 100;

CSS Classes: kebab-case
  className="primary-button"
```

### Tailwind Classes
```
Never abbreviate:
  ✓ className="padding-x-4 padding-y-2"
  ✗ className="px-4 py-2"

Actually, DO abbreviate (Tailwind standard):
  ✓ className="px-4 py-2 bg-primary"
  ✗ className="padding-left-4 padding-right-4"
```

## Common Patterns

### Error Handling
```typescript
try {
  // API call
} catch (error) {
  const message = error instanceof Error ? error.message : 'Unknown error';
  console.error('Error:', message);
  // User feedback
}
```

### Form Validation
```typescript
const validateEmail = (email: string) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
const validatePhone = (phone: string) => /^\d{10,}$/.test(phone.replace(/\D/g, ''));
const validateRequired = (value: string) => value.trim().length > 0;
```

### Loading States
```typescript
const [isLoading, setIsLoading] = useState(false);

const handleSubmit = async () => {
  setIsLoading(true);
  try {
    // API call
  } finally {
    setIsLoading(false);
  }
};
```

## Accessibility Standards

1. **Semantic HTML**
   ```
   ✓ <button>, <nav>, <main>, <section>
   ✗ <div onClick={...}>, <span> for interactive
   ```

2. **ARIA Labels**
   ```
   aria-label="Close menu"
   aria-describedby="error-message"
   aria-expanded={isOpen}
   ```

3. **Focus Management**
   ```
   ✓ Keyboard navigation works
   ✓ Focus visible on interactive elements
   ✓ Tab order is logical
   ```

4. **Color Contrast**
   ```
   ✓ Text on primary: White (high contrast)
   ✓ Text on light: Dark grey (good contrast)
   ✗ Light grey on white (insufficient)
   ```

## Performance Considerations

1. **Code Splitting**
   ```
   Use dynamic imports for large components
   const AdminDashboard = dynamic(() => import('./AdminDashboard'));
   ```

2. **Image Optimization**
   ```
   Use Next.js Image component
   <Image src="/image.jpg" alt="..." width={400} height={300} />
   ```

3. **Memoization**
   ```
   For expensive computations or frequent re-renders
   const memoized = useMemo(() => expensive(), [deps]);
   ```
